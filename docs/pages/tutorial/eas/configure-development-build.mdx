function App({ onBack }) {
  const [profile, setProfile] = useState(() => load('profile', { name: 'Usuário', level: 'iniciante', weight: 70 }));
  const [history, setHistory] = useState(() => load('history', []));
  const [workouts, setWorkouts] = useState(() => load('workouts', []));
  const [selectedGoal, setSelectedGoal] = useState('resistencia');
  const [intensity, setIntensity] = useState('moderado');
  const [duration, setDuration] = useState(30);
  const [currentWorkout, setCurrentWorkout] = useState(null);
  const [quizOpen, setQuizOpen] = useState(false);
  const [quizResult, setQuizResult] = useState(null);

  useEffect(() => save('profile', profile), [profile]);
  useEffect(() => save('history', history), [history]);
  useEffect(() => save('workouts', workouts), [workouts]);

  function generateWorkout(fromQuiz = false) {
    const pool = EXERCISES.filter(e => {
      if (selectedGoal === 'forca') return e.type === 'forca';
      if (selectedGoal === 'resistencia') return e.type === 'cardio' || e.type === 'HIIT';
      if (selectedGoal === 'core') return e.type === 'core' || e.type === 'forca';
      return true;
    });

    const sets = Math.max(3, Math.round(duration / 10));
    const chosen = [];
    for (let i = 0; i < Math.min(5, pool.length); i++) {
      const idx = Math.floor(Math.random() * pool.length);
      const ex = pool[idx];
      const target = ex.metric === 'min' ? Math.max(1, Math.round(duration / 5)) : ex.metric === 'seg' ? 30 + i * 15 : 8 + i * 4;
      chosen.push({ ...ex, sets, target });
    }

    const workout = { id: Date.now(), date: new Date().toISOString(), goal: selectedGoal, intensity, duration, exercises: chosen, recommendedFromQuiz: fromQuiz };
    setCurrentWorkout(workout);
    setWorkouts(prev => [workout, ...prev].slice(0, 50));
  }

  function completeWorkout() {
    if (!currentWorkout) return;
    const entry = { ...currentWorkout, completedAt: new Date().toISOString() };
    setHistory(prev => [entry, ...prev].slice(0, 200));
    setCurrentWorkout(null);
  }

  return (
    <div className="min-h-screen bg-gray-100 p-6">
      <div className="max-w-6xl mx-auto grid grid-cols-1 md:grid-cols-3 gap-6">
        <div className="md:col-span-2 space-y-6">
          <div className="flex items-center justify-between p-4 rounded-2xl bg-white shadow">
            <div>
              <h1 className="text-2xl font-bold">FitFlow</h1>
              <p className="text-sm text-gray-500">App protótipo para melhorar condicionamento físico</p>
            </div>
            <div className="flex gap-2">
              <button className="px-3 py-2 rounded bg-gray-100" onClick={onBack}>Voltar ao site</button>
              <div className="text-right">
                <div className="text-sm">{profile.name}</div>
                <div className="text-xs text-gray-500">{profile.weight} kg · {profile.level}</div>
              </div>
            </div>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
            <div className="p-4 rounded-2xl shadow bg-white">
              <h3 className="text-lg font-semibold">Gerador de treino</h3>
              <div className="mt-3 space-y-2">
                <label className="block">Objetivo</label>
                <select className="w-full p-2 rounded" value={selectedGoal} onChange={e => setSelectedGoal(e.target.value)}>
                  <option value="resistencia">Resistência / Cardio</option>
                  <option value="forca">Força</option>
                  <option value="core">Core / Estabilidade</option>
                </select>

                <label>Intensidade</label>
                <select className="w-full p-2 rounded" value={intensity} onChange={e => setIntensity(e.target.value)}>
                  <option>leve</option>
                  <option>moderado</option>
                  <option>intenso</option>
                </select>

                <label>Duração (min)</label>
                <input type="range" min={10} max={90} value={duration} onChange={e => setDuration(Number(e.target.value))} />
                <div className="flex justify-between text-sm"><span>10</span><span>{duration} min</span><span>90</span></div>

                <div className="flex gap-2 mt-3">
                  <button className="px-4 py-2 rounded-2xl shadow bg-indigo-600 text-white" onClick={() => setQuizOpen(true)}>Fazer quiz rápido</button>
                  <button className="px-4 py-2 rounded-2xl shadow bg-gray-200" onClick={() => { setSelectedGoal('resistencia'); setIntensity('moderado'); setDuration(30); }}>Resetar</button>
                </div>

                <div className="flex gap-2 mt-3">
                  <button className="px-4 py-2 rounded-2xl bg-green-500 text-white" onClick={() => generateWorkout(false)}>Gerar treino</button>
                  <button className="px-4 py-2 rounded-2xl bg-yellow-100" onClick={() => { setProfile(p => ({ ...p, level: p.level === 'iniciante' ? 'intermediario' : 'iniciante' })); }}>Alternar nível</button>
                </div>
              </div>
            </div>

            <div className="p-4 rounded-2xl shadow bg-white">
              <h3 className="text-lg font-semibold">Treino atual</h3>
              {currentWorkout ? (
                <div className="mt-3">
                  <p className="text-sm">Objetivo: <strong>{currentWorkout.goal}</strong> · {currentWorkout.duration} min · {currentWorkout.intensity}</p>
                  <ul className="mt-2 space-y-2">
                    {currentWorkout.exercises.map(ex => (
                      <li key={ex.id} className="p-2 rounded border">{ex.name} — {ex.target} {ex.metric} · {ex.sets} séries</li>
                    ))}
                  </ul>
                  <div className="flex gap-2 mt-3">
                    <button className="px-4 py-2 rounded-2xl bg-green-500 text-white" onClick={completeWorkout}>Concluir</button>
                    <HIITTimer />
                  </div>
                </div>
              ) : (
                <p className="mt-3 text-sm text-gray-600">Nenhum treino gerado. Use o gerador ou faça o quiz rápido.</p>
              )}
            </div>
          </div>

          <div className="p-4 rounded-2xl shadow bg-white">
            <h3 className="text-lg font-semibold">Histórico</h3>
            {history.length === 0 ? <p className="text-sm mt-2 text-gray-600">Nenhum treino concluído ainda.</p> : (
              <div className="mt-3 space-y-2">
                {history.slice(0,6).map(h => (
                  <div key={h.id} className="p-2 rounded border flex justify-between items-center">
                    <div>
                      <div className="text-sm font-medium">{h.goal} · {h.duration} min</div>
                      <div className="text-xs text-gray-500">Concluído: {new Date(h.completedAt).toLocaleString()}</div>
                    </div>
                    <div className="text-sm">Exs: {h.exercises.length}</div>
                  </div>
                ))}
              </div>
            )}
          </div>

        </div>

        <aside className="space-y-6">
          <div className="p-4 rounded-2xl shadow bg-white">
            <h4 className="font-semibold">Progresso</h4>
            <ProgressChart history={history} />
          </div>

          <div className="p-4 rounded-2xl shadow bg-white">
            <h4 className="font-semibold">Planos salvos</h4>
            {workouts.length === 0 ? <p className="text-sm text-gray-500">Nenhum plano salvo.</p> : (
              <ul className="space-y-2 mt-2">
                {workouts.slice(0,6).map(w => (
                  <li key={w.id} className="p-2 rounded border flex justify-between items-center">
                    <div>
                      <div className="text-sm font-medium">{w.goal} · {w.duration} min</div>
                      <div className="text-xs text-gray-500">Gerado: {new Date(w.date).toLocaleDateString()}</div>
                    </div>
                    <div className="flex gap-1">
                      <button className="px-2 py-1 rounded bg-gray-100" onClick={() => setCurrentWorkout(w)}>Abrir</button>
                      <button className="px-2 py-1 rounded bg-red-100" onClick={() => setWorkouts(prev => prev.filter(x => x.id !== w.id))}>Excluir</button>
                    </div>
                  </li>
                ))}
              </ul>
            )}
          </div>

          <div className="p-4 rounded-2xl shadow bg-white">
            <h4 className="font-semibold">Configurações rápidas</h4>
            <div className="mt-2 text-sm text-gray-600">Nível: {profile.level}</div>
          </div>

        </aside>
      </div>

      {quizOpen && <QuizModal onClose={() => setQuizOpen(false)} onSubmit={(result) => { setQuizResult(result); setQuizOpen(false); generateWorkout(true); }} />}
    </div>
  );
}

function ProgressChart({ history }){
  const map = {};
  history.forEach(h => {
    const d = h.completedAt ? new Date(h.completedAt).toLocaleDateString() : new Date(h.date).toLocaleDateString();
    map[d] = (map[d] || 0) + 1;
  });
  const entries = Object.entries(map).sort((a,b) => new Date(a[0]) - new Date(b[0]));
  const chartData = entries.slice(-14).map(([day, count]) => ({ day, count }));

  if (chartData.length === 0) return <p className="text-sm text-gray-500 mt-2">Sem dados para mostrar.</p>;

  return (
    <div style={{ height: 160 }} className="mt-2">
      <ResponsiveContainer width="100%" height="100%">
        <LineChart data={chartData}>
          <CartesianGrid strokeDasharray="3 3" />
          <XAxis dataKey="day" hide />
          <YAxis allowDecimals={false} />
          <Tooltip />
          <Line type="monotone" dataKey="count" stroke="#8884d8" strokeWidth={3} dot={false} />
        </LineChart>
      </ResponsiveContainer>
    </div>
  );
}

function HIITTimer(){
  const [running, setRunning] = useState(false);
  const [secs, setSecs] = useState(30);
  const ref = useRef(null);

  useEffect(() => {
    if (!running) return;
    ref.current = setInterval(() => setSecs(s => s - 1), 1000);
    return () => clearInterval(ref.current);
  }, [running]);

  useEffect(() => {
    if (secs <= 0 && running) {
      clearInterval(ref.current);
      setRunning(false);
      setSecs(30);
    }
  }, [secs, running]);

  return (
    <div className="flex items-center gap-2">
      <div className="text-sm px-3 py-2 rounded bg-gray-100">{secs}s</div>
      <button className="px-3 py-2 rounded bg-indigo-600 text-white flex items-center gap-2" onClick={() => setRunning(r => !r)}>
        {running ? <Pause size={16} /> : <Play size={16} />}
        {running ? 'Pausar' : 'Iniciar'}
      </button>
    </div>
  );
}

function QuizModal({ onClose, onSubmit }){
  const [step, setStep] = useState(0);
  const [answers, setAnswers] = useState({ level: null, freq: null, prefer: null });

  const questions = [
    { key: 'level', q: 'Como você avalia seu condicionamento atual?', opts: ['Iniciante', 'Intermediário', 'Avançado'] },
    { key: 'freq', q: 'Quantos dias por semana você pretende treinar?', opts: ['1-2', '3-4', '5+'] },
    { key: 'prefer', q: 'Você prefere treinos mais curtos e intensos ou longos e moderados?', opts: ['Curtos e intensos', 'Longos e moderados', 'Sem preferência'] }
  ];

  function choose(opt){
    const key = questions[step].key;
    setAnswers(a => ({ ...a, [key]: opt }));
  }

  function next(){
    if (!answers[questions[step].key]) return;
    if (step < questions.length - 1) setStep(s => s + 1);
    else submit();
  }

  function submit(){
    onSubmit({ ...answers });
  }

  return (
    <div className="fixed inset-0 bg-black/40 flex items-center justify-center z-50">
      <div className="bg-white rounded-2xl w-full max-w-md p-6">
        <div className="flex justify-between items-center">
          <h4 className="font-semibold">Quiz rápido ({step+1}/{questions.length})</h4>
          <button onClick={onClose}><X /></button>
        </div>
        <div className="mt-4">
          <div className="font-medium">{questions[step].q}</div>
          <div className="mt-3 grid grid-cols-1 gap-2">
            {questions[step].opts.map(o => (
              <button key={o} onClick={() => choose(o)} className={`w-full text-left p-3 rounded ${answers[questions[step].key] === o ? 'border-2 border-indigo-500' : 'border'}`}>
                {o}
              </button>
            ))}
          </div>
        </div>
        <div className="mt-4 flex justify-end gap-2">
          <button className="px-4 py-2 rounded bg-gray-100" onClick={onClose}>Cancelar</button>
          <button className="px-4 py-2 rounded bg-indigo-600 text-white" onClick={next}>Próximo</button>
        </div>
      </div>
    </div>
  );
}

function ComprarViaPIX({ planId }){
  const [qrCode, setQrCode] = useState(null);
  const [payload, setPayload] = useState(null);
  const [loading, setLoading] = useState(false);

  async function handleBuy(){
    try{
      setLoading(true);
      const res = await fetch('/api/pix/create-charge', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ planId })
      });
      if (!res.ok) throw new Error('Erro ao criar cobrança');
      const data = await res.json();
      setQrCode(data.qrCodeBase64 || null);
      setPayload(data.copyPasteCode || null);
    }catch(e){
      console.error(e);
    }finally{ setLoading(false); }
  }

  return (
    <div className="space-y-4">
      <button onClick={handleBuy} disabled={loading} className="w-full bg-indigo-600 text-white py-3 rounded-lg font-semibold">
        Comprar via PIX (Vitalício)
      </button>
      {qrCode && (
        <div className="text-center space-y-2">
          <img src={`data:image/png;base64,${qrCode}`} alt="QR Code PIX" className="mx-auto w-48 h-48" />
          {payload && <button onClick={() => navigator.clipboard.writeText(payload)} className="text-sm underline text-indigo-500">Copiar código PIX</button>}
        </div>
      )}
    </div>
  );
}

      )}

      <div>
        <div className="text-sm">Envie o comprovante (opcional)</div>
        <input type="file" accept="image/*,application/pdf" onChange={(e) => setFile(e.target.files && e.target.files[0] ? e.target.files[0] : null)} />
        <div className="mt-2 flex gap-2">
          <button onClick={sendProof} className="px-3 py-2 rounded bg-indigo-600 text-white">Enviar Comprovante</button>
          <button onClick={() => setStatus('aguardando_pagamento')} className="px-3 py-2 rounded bg-gray-100">Cancelar</button>
        </div>
      </div>

      <div>
        {status === 'aguardando_pagamento' && <div className="text-sm text-gray-600">Status: Aguardando pagamento</div>}
        {status === 'em_validacao' && <div className="text-sm text-yellow-600">Status: Em validação...</div>}
        {status === 'aprovado' && <div className="text-sm text-green-600">Status: ✅ Pagamento confirmado</div>}
      </div>
    </div>
  );
}
